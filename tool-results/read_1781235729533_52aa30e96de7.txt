   606→// ---- Computed Dashboard Stats ----
   607→export function computeDashboardStats(
   608→  clients: Client[],
   609→  workflows: Workflow[],
   610→  events: WorkflowEvent[]
   611→): DashboardStats {
   612→  const now = new Date();
   613→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
   614→  const dayAgo = new Date(now.getTime() - 24 * 60 * 60 * 1000);
   615→
   616→  const failedRunsThisWeek = events.filter(
   617→    (e) => e.outcome === 'failure' && new Date(e.timestamp) >= weekAgo
   618→  ).length;
   619→
   620→  const silentIssuesThisWeek = events.filter(
   621→    (e) => e.isSilent && new Date(e.timestamp) >= weekAgo
   622→  ).length;
   623→
   624→  const criticalEventsLast24h = events.filter(
   625→    (e) =>
   626→      e.severity === 'critical' &&
   627→      new Date(e.timestamp) >= dayAgo
   628→  ).length;
   629→
   630→  return {
   631→    totalClients: clients.length,
   632→    totalWorkflows: workflows.length,
   633→    healthyWorkflows: workflows.filter((w) => w.status === 'healthy').length,
   634→    warningWorkflows: workflows.filter((w) => w.status === 'warning').length,
   635→    criticalWorkflows: workflows.filter((w) => w.status === 'critical').length,
   636→    failedRunsThisWeek,
   637→    silentIssuesThisWeek,
   638→    criticalEventsLast24h,
   639→    avgClientHealthScore:
   640→      Math.round(
   641→        clients.reduce((sum, c) => sum + c.healthScore, 0) / clients.length
   642→      ),
   643→  };
   644→}
